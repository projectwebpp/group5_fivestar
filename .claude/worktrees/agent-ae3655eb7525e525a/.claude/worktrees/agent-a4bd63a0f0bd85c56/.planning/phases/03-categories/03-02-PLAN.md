```typescript
import apiClient from './client';

export interface Category {
  id: number;
  name: string;
  icon: string;
  color: string;
  user_id: number;
  created_at: string;
  updated_at: string;
}

export interface CategoryPayload {
  name: string;
  icon: string;
  color: string;
}

interface Envelope<T> {
  success: boolean;
  data: T;
  message: string;
}

export const getCategories = () =>
  apiClient.get<Envelope<Category[]>>('/categories');

export const createCategory = (payload: CategoryPayload) =>
  apiClient.post<Envelope<Category>>('/categories', payload);

export const updateCategory = (id: number, payload: CategoryPayload) =>
  apiClient.put<Envelope<Category>>(`/categories/${id}`, payload);

export const deleteCategory = (id: number) =>
  apiClient.delete<Envelope<null>>(`/categories/${id}`);
```

**useEffect on mount:** Call `getCategories()`, set state on success, set pageError on failure, set loading false in finally.

**openCreateModal():** Reset form to defaults (`formName=''`, `formIcon='utensils'`, `formColor='#FF6B6B'`), set `editingCategory=null`, open modal.

**openEditModal(cat):** Pre-fill form with `cat.name`, `cat.icon`, `cat.color`, set `editingCategory=cat`, open modal.

**closeModal():** Set `modalOpen=false`, clear formError.

**handleSubmit(e):** Prevent default, clear formError, set formLoading. Branch on `editingCategory`: if set, call `updateCategory` and replace the category in state array; else call `createCategory` and append. On success, call `closeModal()`. On error, extract `.response.data.message` or fallback to generic message. Always set formLoading=false in finally.

**handleDelete(id):** Set formLoading equivalent is not needed — track delete in-progress separately or disable the button during call. Call `deleteCategory(id)`. On success, remove category from state. On 422, set `deleteErrors[id]` to the exact string `'This category has expenses and cannot be deleted.'` (from UI-SPEC copywriting). On other errors, set generic message. After success or error, clear `confirmingDelete`.
