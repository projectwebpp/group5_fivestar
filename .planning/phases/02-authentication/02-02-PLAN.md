```tsx
import { Navigate } from 'react-router-dom';
import type { ReactElement } from 'react';

interface ProtectedRouteProps {
  children: ReactElement;
}

export default function ProtectedRoute({ children }: ProtectedRouteProps) {
  const token = localStorage.getItem('auth_token');
  if (!token) {
    return <Navigate to="/auth" replace />;
  }
  return children;
}
```

```tsx
import { useState } from 'react';
import type { CSSProperties, FormEvent } from 'react';
import { useNavigate } from 'react-router-dom';
import apiClient from '../api/client';

type Tab = 'login' | 'register';

interface AuthSuccess {
  success: true;
  data: { token: string };
  message: string;
}

interface AxiosErrorShape {
  response?: { data?: { message?: string } };
}

export default function AuthPage() {
  const [tab, setTab] = useState<Tab>('login'); // D-02: login default
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState(''); // D-11
  const [loading, setLoading] = useState(false); // D-09
  const [error, setError] = useState<string | null>(null); // D-08
  const navigate = useNavigate();

  const switchTab = (next: Tab) => {
    setTab(next);
    setError(null);
    setPassword('');
    setConfirmPassword('');
  };

  const handleSubmit = async (e: FormEvent) => {
    e.preventDefault();
    setError(null);
    setLoading(true);
    try {
      const endpoint = tab === 'login' ? '/auth/login' : '/auth/register';
      const body =
        tab === 'login'
          ? { email, password }
          : { email, password, password_confirmation: confirmPassword }; // Pitfall 2
      const res = await apiClient.post<AuthSuccess>(endpoint, body);
      const token = res.data?.data?.token;
      if (!token) {
        setError('Authentication failed: no token returned');
        return;
      }
      localStorage.setItem('auth_token', token); // D-06
      navigate('/'); // D-03
    } catch (err) {
      const msg =
        (err as AxiosErrorShape)?.response?.data?.message ?? 'Something went wrong';
      setError(msg); // D-08
    } finally {
      setLoading(false);
    }
  };

  const tabButtonStyle = (active: boolean): CSSProperties => ({
    fontWeight: active ? 'bold' : 'normal',
    padding: '0.5rem 1rem',
    marginRight: '0.5rem',
    cursor: 'pointer',
  });

  const submitLabel = loading
    ? 'Loading...'
    : tab === 'login'
      ? 'Log In'
      : 'Register';

  return (
    <div style={{ fontFamily: 'sans-serif', padding: '2rem', maxWidth: '420px' }}>
      <h1>Expense Tracker</h1>

      <div style={{ marginBottom: '1rem' }}>
        <button
          type="button"
          onClick={() => switchTab('login')}
          style={tabButtonStyle(tab === 'login')}
        >
          Login
        </button>
        <button
          type="button"
          onClick={() => switchTab('register')}
          style={tabButtonStyle(tab === 'register')}
        >
          Register
        </button>
      </div>

      <form onSubmit={handleSubmit}>
        <div style={{ marginBottom: '0.75rem' }}>
          <label>
            Email
            <br />
            <input
              type="email"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              required
              style={{ width: '100%', padding: '0.5rem' }}
            />
          </label>
        </div>

        <div style={{ marginBottom: '0.75rem' }}>
          <label>
            Password
            <br />
            <input
              type="password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              required
              minLength={tab === 'register' ? 8 : undefined}
              style={{ width: '100%', padding: '0.5rem' }}
            />
          </label>
        </div>

        {tab === 'register' && (
          <div style={{ marginBottom: '0.75rem' }}>
            <label>
              Confirm Password
              <br />
              <input
                type="password"
                value={confirmPassword}
                onChange={(e) => setConfirmPassword(e.target.value)}
                required
                minLength={8}
                style={{ width: '100%', padding: '0.5rem' }}
              />
            </label>
          </div>
        )}

        <button
          type="submit"
          disabled={loading}
          style={{ padding: '0.5rem 1rem', cursor: loading ? 'not-allowed' : 'pointer' }}
        >
          {submitLabel}
        </button>

        {error && (
          <p style={{ color: 'red', marginTop: '0.75rem' }} role="alert">
            {error}
          </p>
        )}
      </form>
    </div>
  );
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser DOM -> localStorage | localStorage is readable by any script running on the same origin (XSS risk). |
| Browser -> Laravel API | HTTPS in production (Vercel + Railway); CORS limits origin via `FRONTEND_URL` env. |
| User input -> React component state | Untrusted email/password text reaches state and is forwarded to API; no rendering as HTML. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-2-09 | Information Disclosure | XSS exfiltrates auth_token from localStorage | accept | localStorage choice locked by D-06 for v1. React's default escaping prevents inline-script injection from form values; we never `dangerouslySetInnerHTML`. CSP not added in v1 (school project). Documented risk, revisited in v2 (httpOnly cookie path). |
| T-2-10 | Spoofing | Client-side ProtectedRoute can be bypassed by editing localStorage | mitigate (defense-in-depth) | The CLIENT-SIDE check is UX only; AUTH-04 is enforced authoritatively by the `jwt.auth` middleware on the backend (Plan 02-01). A user inserting `localStorage.setItem('auth_token','fake')` will pass ProtectedRoute but every API call returns 401, rendering protected pages empty. |
| T-2-11 | Tampering | Forged response (MITM) injects token into localStorage | mitigate | Production traffic uses HTTPS (Railway + Vercel default). Local dev exposes plaintext but no real credentials are used there. |
| T-2-12 | Information Disclosure | Token sent in URL query string or browser history | mitigate | Token is sent in JSON body (POST) or `Authorization: Bearer` header — never in URL. `<Navigate to="/auth" replace />` uses `replace` to avoid history entries from the redirect (and never carries the token in the URL anyway). |
| T-2-13 | Repudiation | Logged-in token persists indefinitely after stolen device access | accept | JWT TTL is 24h (D-07) — bounded exposure window. No "logout from all devices" UI in v1 (deferred). Acceptable for school-project scope. |
| T-2-14 | Tampering | Form input bypassing min-length (e.g., curl to backend) | mitigate (server side) | HTML `minLength={8}` is UX only; backend `password|required|min:8|confirmed` (Plan 02-01) is the authoritative gate. Verified in AuthTest::test_register_rejects_short_password. |

**Block on high severity:** T-2-09 is documented and accepted at planner level (D-06 lock). T-2-10..T-2-14 are mitigated. No high-severity unhandled threats. Any future addition of a logout UI inherits T-2-13 mitigation by clearing the token.
