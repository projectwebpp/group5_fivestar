       ```php
       namespace App\Http\Controllers\Api;
       use App\Http\Controllers\Controller;
       use App\Models\Expense;
       use App\Http\Requests\StoreExpenseRequest;
       use App\Http\Requests\UpdateExpenseRequest;
       use Illuminate\Http\Request;
       use Illuminate\Support\Facades\Auth;

       class ExpenseController extends Controller {
         public function index(Request $r) {
           $q = Expense::query()->where('user_id', Auth::id());
           if ($r->filled('category_id')) $q->where('category_id', $r->integer('category_id'));
           if ($r->filled('date_from'))  $q->whereDate('expense_date', '>=', $r->date('date_from'));
           if ($r->filled('date_to'))    $q->whereDate('expense_date', '<=', $r->date('date_to'));
           if ($r->filled('amount_min')) $q->where('amount', '>=', (float)$r->input('amount_min'));
           if ($r->filled('amount_max')) $q->where('amount', '<=', (float)$r->input('amount_max'));
           $perPage = 10; // D-07 fixed
           $page = $q->orderBy('expense_date','desc')->orderBy('id','desc')->paginate($perPage);
           return response()->success([
             'items' => $page->getCollection()->map(fn($e)=>$this->shape($e))->all(),
             'meta'  => ['page'=>$page->currentPage(),'per_page'=>$perPage,'total'=>$page->total(),'last_page'=>$page->lastPage()],
           ], 'OK');
         }
         public function store(StoreExpenseRequest $r) {
           $data = $r->validated();
           $e = Expense::create([
             'user_id' => Auth::id(),
             'amount' => $data['amount'],
             'currency' => 'THB', // D-13 silent server-set
             'category_id' => $data['category_id'],
             'description' => $data['description'],
             'expense_date' => $data['date'],
             'notes' => $data['notes'] ?? null,
           ]);
           return response()->success($this->shape($e), 'Created', 201);
         }
         public function show(string $id) {
           $e = Expense::where('user_id', Auth::id())->find($id);
           if (!$e) return response()->error('Expense not found', [], 404);
           return response()->success($this->shape($e), 'OK');
         }
         public function update(UpdateExpenseRequest $r, string $id) {
           $e = Expense::where('user_id', Auth::id())->find($id);
           if (!$e) return response()->error('Expense not found', [], 404);
           $data = $r->validated();
           if (isset($data['date'])) { $data['expense_date'] = $data['date']; unset($data['date']); }
           $e->fill($data)->save();
           return response()->success($this->shape($e->fresh()), 'Updated');
         }
         public function destroy(string $id) {
           $e = Expense::where('user_id', Auth::id())->find($id);
           if (!$e) return response()->error('Expense not found', [], 404);
           $e->delete();
           return response()->success(['id' => (int)$id], 'Deleted');
         }
         private function shape(Expense $e): array {
           return [
             'id' => $e->id,
             'amount' => (float)$e->amount,
             'currency' => $e->currency,
             'category_id' => $e->category_id,
             'description' => $e->description,
             'date' => $e->expense_date?->format('Y-m-d'),
             'notes' => $e->notes,
             'created_at' => $e->created_at?->toIso8601String(),
             'updated_at' => $e->updated_at?->toIso8601String(),
           ];
         }
       }
       ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser → /api/expenses* | Untrusted JSON payloads cross here; JWT must validate |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Spoofing | All /api/expenses* routes | mitigate | Route group wrapped in `auth:api` middleware; tested by `test_endpoints_require_jwt` |
| T-04-02 | Tampering | StoreExpenseRequest amount field | mitigate | `numeric` + `gt:0` + regex `^\d+(\.\d{1,2})?$` ; backend stores currency='THB' regardless of client input |
| T-04-03 | Information Disclosure (IDOR) | show/update/destroy on /api/expenses/{id} | mitigate | Every query starts `Expense::where('user_id', Auth::id())`; non-owned rows return 404 not 403 (don't leak existence); tested by `test_show_returns_404_when_owned_by_other_user` and `test_delete_404_when_other_user_owns` |
| T-04-04 | Tampering (mass assignment) | Expense model fillable | mitigate | Fillable excludes `user_id` setter from request — controller sets `user_id => Auth::id()` server-side; currency hardcoded server-side per D-13 |
| T-04-05 | Denial of Service | GET /api/expenses unbounded | mitigate | Page size pinned to 10 (D-07); no client-controlled per_page param |
| T-04-06 | Injection | filter query params | mitigate | Eloquent query builder parameterizes all values; `whereDate` and `where` bind safely |
| T-04-07 | Repudiation | who created/edited an expense | accept | created_at/updated_at timestamps logged; user_id immutable post-create; full audit log deferred |
| T-04-08 | Elevation of Privilege | category_id pointing to another user's category | mitigate | `exists:categories,id` rule; Phase 3 will add per-user scoping if needed — for v1, predefined categories are global so cross-user category leak is not a concern |
</threat_model>
